use m68k::{AddressBus, CpuCore, CpuType, FastMem};
use std::time::Instant;
const BASE:u32=0x200000;
const WARM:usize=1_048_576;
const COUNT:usize=67_108_864;
const PROGRAM:[u8;8]=[0x20,0x10,0x52,0x80,0x20,0x80,0x60,0xf8];
struct Flat(Vec<u8>);
impl AddressBus for Flat {
    fn read_byte(&mut self, a: u32) -> u8 {
        self.0[(a - BASE) as usize]
    }
    fn read_word(&mut self, a: u32) -> u16 {
        u16::from_be_bytes(self.0[(a - BASE) as usize..][..2].try_into().unwrap())
    }
    fn read_long(&mut self, a: u32) -> u32 {
        u32::from_be_bytes(self.0[(a - BASE) as usize..][..4].try_into().unwrap())
    }
    fn write_byte(&mut self, a: u32, v: u8) {
        self.0[(a - BASE) as usize] = v;
    }
    fn write_word(&mut self, a: u32, v: u16) {
        self.0[(a - BASE) as usize..][..2].copy_from_slice(&v.to_be_bytes());
    }
    fn write_long(&mut self, a: u32, v: u32) {
        self.0[(a - BASE) as usize..][..4].copy_from_slice(&v.to_be_bytes());
    }
    fn fast_mem(&mut self) -> Option<FastMem> {
        Some(FastMem {
            ptr: self.0.as_mut_ptr(),
            base: BASE,
            len: self.0.len() as u32,
        })
    }
}

fn main(){
 for trial in 0..5 { for batch in [64usize,256,1024,4096] {
  let mut mem=Flat(vec![0;65536]);
  mem.0[0x100..0x108].copy_from_slice(&PROGRAM);
  let mut cpu=Box::new(CpuCore::new());cpu.set_cpu_type(CpuType::M68040);cpu.pc=BASE+0x100;cpu.set_a(0,BASE+0x800);
  for _ in 0..WARM/batch { assert_eq!(cpu.run_batch(&mut mem,batch as u32,&[]).instructions,batch as u32); }
  let before=cpu.d(0);let start=Instant::now();
  for _ in 0..COUNT/batch { assert_eq!(cpu.run_batch(&mut mem,batch as u32,&[]).instructions,batch as u32); }
  let elapsed=start.elapsed().as_secs_f64();assert_eq!(cpu.d(0)-before,(COUNT/4)as u32);
  println!("{trial},{batch},{elapsed:.6},{:.2}",COUNT as f64/elapsed/1e6);
 }}
}
